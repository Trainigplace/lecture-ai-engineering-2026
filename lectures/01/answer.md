def main() -> None:
    client = genai.Client(api_key=get_api_key())
    chat = client.chats.create(model=MODEL)  # client の直後に一度だけ作る

    while True:
        question = input("質問（終了は exit）> ").strip()
        if question.lower() == "exit":
            break
        if not question:
            continue

        response = chat.send_message(question)
        print(response.text)
